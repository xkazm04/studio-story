```typescript
export type IntentType = 'compose' | 'manipulate' | 'navigate' | 'query' | 'system';
export type IntentSource = 'click' | 'keyboard' | 'drag' | 'menu' | 'llm' | 'slash-command';

export interface Intent<T extends IntentType = IntentType> {
  id: string;
  type: T;
  payload: IntentPayloadMap[T];
  source: IntentSource;
  timestamp: number;
}

export type IntentResult =
  | { status: 'resolved'; patches: Operation[]; origin: PatchOrigin; description: string }
  | { status: 'needs-llm'; intent: Intent }
  | { status: 'error'; error: string };

export interface IntentBus {
  dispatch(intent: Intent): IntentResult;
  subscribe(listener: (event: IntentEvent) => void): () => void;
  getSnapshot(): string;
}
```

       Create a standalone `parseTextToIntent` function that maps common natural language patterns to Intent objects. Cover these patterns:
       - "undo" / "redo" -> system intent
       - "show [panel-type]" / "open [panel-type]" -> compose intent (action: 'open', panelType from known panel types)
       - "close [panel-type]" / "hide [panel-type]" -> compose intent (action: 'close', panelType)
       - "layout [template]" / "switch to [template]" -> compose intent (action: 'set-layout', template)
       - Return null if no pattern matches (falls through to LLM)

       Create a `createInteractionContext()` factory that returns an InteractionContext with unique ID (crypto.randomUUID), empty fragments, state 'accumulating', and createdAt timestamp.

       Create the `useMultimodalInput` hook that:
       - Takes optional `liveClientRef: React.RefObject<GeminiLiveClient | null>` parameter
       - Maintains `interactionContext` in a ref (not state -- avoid re-renders on each fragment)
       - Has `INTERACTION_TIMEOUT_MS = 10000` constant
       - `handleTranscription(text: string)`: Adds voice fragment to context. Calls `parseTextToIntent(text)`. If parseable, dispatches via `useIntent().dispatch()`. If dispatch returns `{ status: 'needs-llm' }` or parseTextToIntent returns null, the voice audio is already being processed by Gemini Live so no additional action needed (Gemini will respond with tool calls). Logs the transcription to agentStore as a user message.
       - `handleTextInput(text: string)`: Adds text fragment to context. Same resolution path as handleTranscription. If needs-llm, sends text to `liveClientRef.current?.send(text)` (or falls back to HTTP advisor if no live client).
       - `addClickContext(entityType: string, entityId: string)`: Adds click fragment to context with content `"${entityType}:${entityId}"`.
       - `getInteractionContext()`: Returns current context or null if expired.
       - `clearInteractionContext()`: Resets context.
       - Uses `setTimeout` to expire context after `INTERACTION_TIMEOUT_MS` of no new fragments. Each new fragment resets the timer.

       The hook should use `useIntent()` from `@dzin/core` for dispatching. Import `useIntent` from the dzin package.

The hook should use `'llm'` as the source for voice intents (closest match, since voice goes through Gemini LLM) until the type is extended.
